# invalidate every number downstream. `discovery.py` imports it at module level
# and `impact.py` imports `discovery`, so its absence blocks the whole stage
# rather than degrading it.
#
# Note the import name differs from the distribution name:
#     pip install causal-learn   ->   import causallearn
causal-learn>=0.1.3.0
